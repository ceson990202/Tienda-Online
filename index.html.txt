        // ========== CONFIGURACIÓN ==========

        const ADMIN_PASSWORD = 'admin123';
        const WHATSAPP_NUMBER = '5491123456789';

        // ========== PRODUCTOS INICIALES ==========
        let products = [
            {
                id: 1,
                title: "Laptop HP Pavilion",

            },
            {
                id: 2,

                description: "Pantalla 6.5 pulgadas, 128GB, Cámara 48MP, Batería 5000mAh",
                price: 699.99,

                category: "electronics",

            },
            {

                description: "Cancelación de ruido, Bluetooth 5.0, 30 horas de batería",

        ];

        let nextId = 4;
        let isAdmin = false;

        // ========== AUTENTICACIÓN ==========
        function login() {
            const password = document.getElementById('loginPassword').value;
            if (password === ADMIN_PASSWORD) {
                isAdmin = true;
                document.getElementById('loginModal').classList.remove('active');
                document.getElementById('loginPassword').value = '';
                document.getElementById('loginError').style.display = 'none';
                document.getElementById('adminToggleBtn').textContent = '👑 Admin (activo)';
                document.getElementById('adminToggleBtn').classList.add('active');
                showAdminPanel();
                showToast('✅ Sesión iniciada como administrador');
            } else {
                document.getElementById('loginError').style.display = 'block';
                document.getElementById('loginPassword').value = '';
                document.getElementById('loginPassword').focus();
            }
        }

        function logout() {
            isAdmin = false;
            document.getElementById('adminPanel').classList.remove('active');
            document.getElementById('adminToggleBtn').textContent = '🔐 Admin';
            document.getElementById('adminToggleBtn').classList.remove('active');
            showToast('👋 Sesión cerrada');
        }

        // ========== MOSTRAR/OCULTAR PANEL ADMIN ==========
        document.getElementById('adminToggleBtn').addEventListener('click', () => {
            if (isAdmin) {
                // Si ya está logueado, mostrar/ocultar panel
                document.getElementById('adminPanel').classList.toggle('active');
                if (document.getElementById('adminPanel').classList.contains('active')) {
                    renderAdminTable();
                }
            } else {
                // Mostrar modal de login
                document.getElementById('loginModal').classList.add('active');
                document.getElementById('loginPassword').focus();
            }
        });

        // Cerrar modal de login con ESC
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Escape') {
                document.getElementById('loginModal').classList.remove('active');
                document.getElementById('productModal').classList.remove('active');
            }
        });

        function showAdminPanel() {
            document.getElementById('adminPanel').classList.add('active');
            renderAdminTable();
        }

        // ========== RENDERIZAR TABLA DE ADMIN ==========
        function renderAdminTable() {
            const container = document.getElementById('adminTableContainer');
            
            if (products.length === 0) {
                container.innerHTML = `
                    <div style="text-align:center;padding:30px;color:#999;">
                        <p>📦 No hay productos agregados</p>
                        <button class="btn-add" onclick="showAddProductForm()" style="margin-top:15px;">➕ Agregar primer producto</button>
                    </div>
                `;
                return;
            }

            let html = `
                <div style="overflow-x:auto;">
                    <table class="admin-table">
                        <thead>
                            <tr>
                                <th>Imagen</th>
                                <th>Producto</th>
                                <th>Precio</th>
                                <th>Stock</th>
                                <th>Categoría</th>
                                <th>Acciones</th>
                            </tr>
                        </thead>
                        <tbody>
            `;

            products.forEach(product => {
                html += `
                    <tr>
                        <td><img src="${product.image}" alt="${product.title}" class="product-thumb" onerror="this.src='https://via.placeholder.com/50/cccccc/666666?text=No'"></td>
                        <td><strong>${product.title}</strong><br><small>${product.description.substring(0, 50)}...</small></td>
                        <td>$${product.price.toFixed(2)}</td>
                        <td><span style="color:${product.stock > 0 ? '#2e7d32' : '#c62828'}">${product.stock}</span></td>
                        <td>${product.category || 'Sin categoría'}</td>
                        <td>
                            <button class="btn-edit" onclick="editProduct(${product.id})">✏️ Editar</button>
                            <button class="btn-delete" onclick="deleteProduct(${product.id})">🗑️ Eliminar</button>
                        </td>
                    </tr>
                `;
            });

            html += `
                        </tbody>
                    </table>
                </div>
                <div style="margin-top:15px;color:#999;font-size:0.9em;">
                    Total: ${products.length} productos
                </div>
            `;

            container.innerHTML = html;
        }

        // ========== CRUD DE PRODUCTOS ==========
        function showAddProductForm() {
            if (!isAdmin) {
                showToast('⚠️ Debes iniciar sesión como administrador');
                return;
            }
            document.getElementById('productModalTitle').textContent = '📝 Agregar Producto';
            document.getElementById('productTitle').value = '';
            document.getElementById('productDescription').value = '';
            document.getElementById('productPrice').value = '';
            document.getElementById('productImage').value = '';
            document.getElementById('productCategory').value = '';
            document.getElementById('productStock').value = '10';
            document.getElementById('editingProductId').value = '';
            document.getElementById('productModal').classList.add('active');
        }

        function editProduct(id) {
            if (!isAdmin) {
                showToast('⚠️ Debes iniciar sesión como administrador');
                return;
            }
            const product = products.find(p => p.id === id);
            if (!product) return;

            document.getElementById('productModalTitle').textContent = '✏️ Editar Producto';
            document.getElementById('productTitle').value = product.title;
            document.getElementById('productDescription').value = product.description;
            document.getElementById('productPrice').value = product.price;
            document.getElementById('productImage').value = product.image;
            document.getElementById('productCategory').value = product.category || '';
            document.getElementById('productStock').value = product.stock;
            document.getElementById('editingProductId').value = id;
            document.getElementById('productModal').classList.add('active');
        }

        function saveProduct() {
            const id = document.getElementById('editingProductId').value;
            const title = document.getElementById('productTitle').value.trim();
            const description = document.getElementById('productDescription').value.trim();
            const price = parseFloat(document.getElementById('productPrice').value);
            const image = document.getElementById('productImage').value.trim() || 'https://via.placeholder.com/400x250/cccccc/666666?text=Sin+imagen';
            const category = document.getElementById('productCategory').value.trim();
            const stock = parseInt(document.getElementById('productStock').value) || 0;

            if (!title || !description || isNaN(price) || price <= 0) {
                showToast('⚠️ Por favor completa todos los campos correctamente');
                return;
            }

            if (id) {
                // Editar producto existente
                const index = products.findIndex(p => p.id === parseInt(id));
                if (index !== -1) {
                    products[index] = {
                        ...products[index],
                        title,
                        description,
                        price,
                        image,
                        category,
                        stock
                    };
                    showToast('✅ Producto actualizado correctamente');
                }
            } else {
                // Agregar nuevo producto
                const newProduct = {
                    id: nextId++,
                    title,
                    description,
                    price,
                    image,
                    category,
                    stock
                };
                products.push(newProduct);
                showToast('✅ Producto agregado correctamente');
            }

            closeProductModal();
            renderAdminTable();
            renderProducts();
        }

        function deleteProduct(id) {
            if (!isAdmin) return;
            if (confirm('¿Estás seguro de eliminar este producto?')) {
                products = products.filter(p => p.id !== id);
                renderAdminTable();
                renderProducts();
                showToast('🗑️ Producto eliminado');
            }
        }

        function closeProductModal() {
            document.getElementById('productModal').classList.remove('active');
        }

        // ========== RENDERIZAR PRODUCTOS (público) ==========
        function renderProducts() {
            const container = document.getElementById('productsContainer');
            container.innerHTML = '';

            if (products.length === 0) {
                container.innerHTML = `
                    <div style="grid-column: 1/-1; text-align:center; padding: 50px; color: #999;">
                        <h3>📦 No hay productos disponibles</h3>
                        <p>Vuelve pronto para ver nuestros productos</p>
                    </div>
                `;
                return;
            }

            products.forEach(product => {
                const card = document.createElement('div');
                card.className = 'product-card';
                
                card.innerHTML = `
                    <img src="${product.image}" alt="${product.title}" class="product-image"
                         onerror="this.onerror=null; this.src='https://via.placeholder.com/400x250/cccccc/666666?text=Imagen+no+disponible'">
                    <div class="product-info">
                        <h3 class="product-title">${product.title}</h3>
                        <p class="product-description">${product.description}</p>
                        <p class="product-price">$${product.price.toFixed(2)}</p>
                        <div style="margin-bottom:10px;font-size:0.9em;color:${product.stock > 0 ? '#2e7d32' : '#c62828'};">
                            ${product.stock > 0 ? `✅ ${product.stock} unidades disponibles` : '❌ Agotado'}
                        </div>
                        <button class="buy-button" onclick="buyProduct(${product.id})" ${product.stock <= 0 ? 'disabled style="opacity:0.5;cursor:not-allowed;"' : ''}>
                            📱 Comprar por WhatsApp
                        </button>
                    </div>
                `;
                
                container.appendChild(card);
            });
        }

        // ========== COMPRAR POR WHATSAPP ==========
        function buyProduct(productId) {
            const product = products.find(p => p.id === productId);
            if (!product) return;

            if (product.stock <= 0) {
                showToast('❌ Producto agotado');
                return;
            }

            const message = `¡Hola! Quiero comprar el siguiente producto:
            \n📦 *${product.title}*
            \n📝 *Descripción:* ${product.description}
            \n💰 *Precio:* $${product.price.toFixed(2)}
            \n\n¿Podrían darme más información sobre disponibilidad?`;

            const encodedMessage = encodeURIComponent(message);
            window.open(`https://wa.me/${WHATSAPP_NUMBER}?text=${encodedMessage}`, '_blank');
        }

        // ========== TOAST NOTIFICATIONS ==========
        function showToast(message) {
            const toast = document.getElementById('toast');
            toast.textContent = message;
            toast.classList.add('show');
            
            clearTimeout(toast.timeout);
            toast.timeout = setTimeout(() => {
                toast.classList.remove('show');
            }, 3000);
        }

        // ========== INICIALIZAR ==========
        renderProducts();
        showToast('🚀 Tienda cargada correctamente');

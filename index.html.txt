        function handleImageUpload(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                const imageData = e.target.result;
                document.getElementById('productImageData').value = imageData;
                document.getElementById('productImage').value = ''; // Limpiar URL
                document.getElementById('imagePreview').src = imageData;
                document.getElementById('imagePreview').style.display = 'block';
                showToast('✅ Imagen cargada correctamente');
            };
            reader.readAsDataURL(file);
        }

        function formatPrice(priceUSD) {
            const rate = EXCHANGE_RATES[currentCurrency] || 1;
            const converted = priceUSD * rate;
            const symbol = CURRENCY_SYMBOLS[currentCurrency] || '$';

            let formatted;
            if (currentCurrency === 'EUR') {
                formatted = `${symbol}${converted.toFixed(2)}`;
            } else if (currentCurrency === 'CUP_EFECTIVO' || currentCurrency === 'CUP_TRANSFERENCIA') {
                formatted = `${symbol}${converted.toFixed(0)}`;
            } else {
                formatted = `${symbol}${converted.toFixed(2)}`;
            }

            return formatted;
        }

        function renderProducts() {
            const container = document.getElementById('productsContainer');
            if (!container) return;

            container.innerHTML = '';

            const searchTerm = document.getElementById('searchInput')?.value.toLowerCase() || '';

            const visibleProducts = products.filter(p => p.visible !== false);
            const filtered = visibleProducts.filter(p => {
                return p.title.toLowerCase().includes(searchTerm) ||
                    p.description.toLowerCase().includes(searchTerm);
            });

            if (filtered.length === 0) {
                container.innerHTML = `
                    <div style="grid-column:1/-1;text-align:center;padding:50px;color:#999;">
                        <h3>🔍 No se encontraron productos</h3>
                    </div>
                `;
                return;
            }

            filtered.forEach(product => {
                const card = document.createElement('div');
                card.className = 'product-card';

                const stars = '⭐'.repeat(Math.floor(product.rating)) + '☆'.repeat(5 - Math.floor(product.rating));

                let stockClass = 'in-stock';
                let stockText = `✅ ${product.stock} unidades`;
                if (product.stock === 0) {
                    stockClass = 'out-of-stock';
                    stockText = '❌ Agotado';
                } else if (product.stock < 5) {
                    stockClass = 'low-stock';
                    stockText = `⚠️ Últimas ${product.stock}`;
                }

                const guaranteeHtml = product.guarantee ?
                    `<span class="guarantee-badge">🛡️ Con garantía</span>` :
                    '';

                card.innerHTML = `
                    <div class="product-image-container">
                        <img src="${product.image}" alt="${product.title}" class="product-image"
                             onerror="this.src='https://via.placeholder.com/400x400/cccccc/666666?text=Sin+imagen'">
                    </div>
                    <div class="product-info">
                        <div class="product-title">${product.title}</div>
                        <div class="product-rating">
                            <span class="stars">${stars}</span>
                            <span class="reviews">(${product.reviews} reseñas)</span>
                        </div>
                        <div class="product-price-container">
                            <span class="product-price">${formatPrice(product.price)}</span>
                        </div>
                        <div class="product-description">${product.description}</div>
                        <div class="product-meta">
                            <span class="stock-status ${stockClass}">${stockText}</span>
                            ${guaranteeHtml}
                        </div>
                        <div class="product-buttons">
                            <button class="btn-whatsapp" onclick="buyProductWhatsApp(${product.id})" 
                                    ${product.stock === 0 ? 'disabled' : ''}>
                                📱 Contactar por WhatsApp
                            </button>
                        </div>
                    </div>
                `;

                container.appendChild(card);
            });
        }

        function sendWhatsApp(product) {
            const priceFormatted = formatPrice(product.price);
            const message = `¡Hola! Quiero comprar el siguiente producto:
            \n📦 *${product.title}*
            \n📝 *Descripción:* ${product.description}
            \n💰 *Precio:* ${priceFormatted} (${CURRENCY_NAMES[currentCurrency]})
            \n🛡️ *Garantía:* ${product.guarantee ? 'Sí, ' + product.guaranteeText : 'No'}

            \n¿Podrían darme más información sobre disponibilidad y formas de pago?`;

            const encodedMessage = encodeURIComponent(message);
            const phoneNumber = WHATSAPP_NUMBER.replace(/[+\s-]/g, '');
            window.open(`https://wa.me/${phoneNumber}?text=${encodedMessage}`, '_blank');
        }

        function saveProduct() {
            const id = document.getElementById('editingProductId').value;
            const title = document.getElementById('productTitle').value.trim();
            const description = document.getElementById('productDescription').value.trim();
            const price = parseFloat(document.getElementById('productPrice').value);
            const image = getProductImage();
            const imageData = document.getElementById('productImageData').value;
            const category = document.getElementById('productCategory').value.trim();
            const stock = parseInt(document.getElementById('productStock').value) || 0;
            const rating = parseFloat(document.getElementById('productRating').value) || 4.5;
            const reviews = parseInt(document.getElementById('productReviews').value) || 0;
            const guarantee = document.getElementById('productGuarantee').checked;
            const guaranteeText = document.getElementById('productGuaranteeText').value.trim();
            const visible = document.getElementById('productVisible').checked;

            if (!title || !description || isNaN(price) || price <= 0) {
                showToast('⚠️ Por favor completa todos los campos correctamente');
                return;
            }

            if (id) {
                const index = products.findIndex(p => p.id === parseInt(id));
                if (index !== -1) {
                    products[index] = {
                        ...products[index],
                        title,
                        description,
                        price,
                        image,
                        imageData: imageData || null,
                        category,
                        stock,
                        rating,
                        reviews,
                        guarantee,
                        guaranteeText: guarantee ? guaranteeText : '',
                        visible
                    };
                    showToast('✅ Producto actualizado correctamente');
                }
            } else {
                const newProduct = {
                    id: nextId++,
                    title,
                    description,
                    price,
                    image,
                    imageData: imageData || null,
                    category,
                    stock,
                    rating,
                    reviews,
                    guarantee,
                    guaranteeText: guarantee ? guaranteeText : '',
                    visible
                };
                products.push(newProduct);
                showToast('✅ Producto agregado correctamente');
            }

            closeProductModal();
            renderAdminTable();
            renderProducts();
        }

        function deleteProduct(id) {
            if (!isAdmin) return;

                products = products.filter(p => p.id !== id);
                renderAdminTable();
                renderProducts();
                showToast('🗑️ Producto eliminado');
            }
        }

        function saveFaq() {
            const index = parseInt(document.getElementById('editingFaqIndex').value);
            const title = document.getElementById('faqTitle').value.trim();
            const content = document.getElementById('faqContent').value.trim();

            if (!title || !content) {
                showToast('⚠️ Por favor completa todos los campos');
                return;
            }

            if (index >= 0 && index < faqItems.length) {
                faqItems[index] = { title, content };
                showToast('✅ FAQ actualizada correctamente');
            } else {
                faqItems.push({ title, content });
                showToast('✅ FAQ agregada correctamente');
            }

            closeFaqModal();
            renderFAQ();
        }

            document.getElementById('faqModal').classList.remove('active');
        }

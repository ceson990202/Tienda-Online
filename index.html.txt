        async function syncToGitHub() {
            if (isSyncing) return;
            
            if (!GITHUB_TOKEN || !GITHUB_USER || !GITHUB_REPO) {
                showToast('⚠️ Configura primero GitHub (Token, Usuario y Repositorio)');
                return;
            }

            isSyncing = true;
            const syncBtn = document.getElementById('syncBtn');
            syncBtn.disabled = true;
            syncBtn.textContent = '⏳ Subiendo...';
            updateGitHubStatus('loading', '⏳ Subiendo datos...');

            try {
                const data = getGitHubData();
                const content = btoa(unescape(encodeURIComponent(JSON.stringify(data, null, 2))));

                // Intentar obtener el archivo actual (para obtener el SHA)
                let sha = null;
                try {
                    const checkResponse = await fetch(
                        `https://api.github.com/repos/${GITHUB_USER}/${GITHUB_REPO}/contents/${GITHUB_FILE}`,
                        { headers: { 'Authorization': `token ${GITHUB_TOKEN}` } }
                    );
                    if (checkResponse.ok) {
                        const fileData = await checkResponse.json();
                        sha = fileData.sha;
                    }
                } catch (e) {}

                // Crear o actualizar el archivo
                const response = await fetch(
                    `https://api.github.com/repos/${GITHUB_USER}/${GITHUB_REPO}/contents/${GITHUB_FILE}`,
                    {
                        method: 'PUT',
                        headers: {
                            'Authorization': `token ${GITHUB_TOKEN}`,
                            'Content-Type': 'application/json'
                        },
                        body: JSON.stringify({
                            message: 'Actualización de datos de la tienda',
                            content: content,
                            sha: sha || undefined
                        })
                    }
                );

                if (!response.ok) {
                    const error = await response.json();
                    throw new Error(error.message || 'Error al subir');
                }

                updateGitHubStatus('sync', '✅ Datos subidos a GitHub');
                showToast('✅ Datos guardados en GitHub correctamente');
                console.log('☁️ Datos subidos a GitHub');

            } catch (error) {
                console.error('Error al sincronizar:', error);
                updateGitHubStatus('unsync', '❌ Error: ' + error.message);
                showToast('❌ Error al subir: ' + error.message);
            }

            isSyncing = false;
            syncBtn.disabled = false;
            syncBtn.textContent = '💾 Guardar en GitHub';
        }

        async function loadFromGitHub() {
            if (!GITHUB_TOKEN || !GITHUB_USER || !GITHUB_REPO) {
                showToast('⚠️ Configura primero GitHub');
                return;
            }

            updateGitHubStatus('loading', '⏳ Cargando desde GitHub...');

            try {
                const response = await fetch(
                    `https://api.github.com/repos/${GITHUB_USER}/${GITHUB_REPO}/contents/${GITHUB_FILE}`,
                    { headers: { 'Authorization': `token ${GITHUB_TOKEN}` } }
                );

                if (!response.ok) {
                    throw new Error('Archivo no encontrado en GitHub');
                }

                const fileData = await response.json();
                const content = JSON.parse(decodeURIComponent(escape(atob(fileData.content))));
                
                if (content && content.products) {
                    products = content.products;
                    nextId = content.nextId || products.length + 1;
                    if (content.faqItems) faqItems = content.faqItems;
                    if (content.paymentMethods) paymentMethods = content.paymentMethods;
                    if (content.whatsappNumber) {
                        WHATSAPP_NUMBER = content.whatsappNumber;
                        document.getElementById('whatsappNumber').value = WHATSAPP_NUMBER;
                        document.getElementById('footerWhatsApp').textContent = '+' + WHATSAPP_NUMBER;
                    }
                    
                    isCloudDataLoaded = true;
                    renderProducts();
                    renderAdminTable();
                    renderFAQ();
                    updateGitHubStatus('sync', '✅ Datos cargados desde GitHub');
                    showToast('✅ Datos cargados desde GitHub correctamente');
                    console.log('☁️ Datos cargados desde GitHub');
                }

            } catch (error) {
                console.warn('⚠️ No se pudieron cargar datos desde GitHub:', error);
                updateGitHubStatus('unsync', '⚠️ No se pudo cargar: ' + error.message);
                showToast('⚠️ No se pudo cargar: ' + error.message);
            }
        }

        function formatPrice(priceUSD) {
            const rate = EXCHANGE_RATES[currentCurrency] || 1;
            const converted = priceUSD * rate;
            const symbol = CURRENCY_SYMBOLS[currentCurrency] || '$';

            let formatted;
            if (currentCurrency === 'EUR') {
                formatted = `${symbol}${converted.toFixed(2)}`;
            } else if (currentCurrency === 'CUP_EFECTIVO' || currentCurrency === 'CUP_TRANSFERENCIA') {
                formatted = `${symbol}${converted.toFixed(0)}`;
            } else {
                formatted = `${symbol}${converted.toFixed(2)}`;
            }

            return formatted;
        }

                            </button>
                        </td>
                        <td>
                            <button class="btn-edit" onclick="editProduct(${product.id})">✏️ Editar</button>
                            <button class="btn-delete" onclick="deleteProduct(${product.id})">🗑️ Eliminar</button>
                        </td>
                    </tr>
                `;
            });

            html += `
                    </tbody>
                </table>
                <div class="table-footer">
                    <span>Total: ${products.length} productos</span>
                    <span>${isCloudDataLoaded ? '☁️ Datos desde GitHub' : '💻 Datos locales'}</span>
                </div>
            `;

            container.innerHTML = html;
        }

        function buyProductWhatsApp(productId) {
            const product = products.find(p => p.id === productId);
            if (!product) {
                showToast('❌ Producto no encontrado');
                return;
            }

            if (product.stock === 0) {
                showToast('❌ Producto agotado');
                return;
            }

            if (product.guarantee && product.guaranteeText) {
                selectedProductId = productId;
                guaranteeCallback = 'whatsapp';
                showGuaranteeModal(product);
                return;
            }

            openWhatsApp(product);
        }

            if (!isAdmin) {

            const product = products.find(p => p.id === id);
            if (!product) return;

            document.getElementById('productTitle').value = product.title;
            document.getElementById('productDescription').value = product.description;
            document.getElementById('productPrice').value = product.price;
            document.getElementById('productImage').value = product.image;
            document.getElementById('productImageData').value = product.imageData || '';
            document.getElementById('productCategory').value = product.category || '';
            document.getElementById('productStock').value = product.stock;
            document.getElementById('productRating').value = product.rating;
            document.getElementById('productReviews').value = product.reviews;
            document.getElementById('productGuarantee').checked = product.guarantee;
            document.getElementById('productGuaranteeText').value = product.guaranteeText || '';
            document.getElementById('guaranteeFields').style.display = product.guarantee ? 'block' : 'none';
            document.getElementById('productVisible').checked = product.visible !== false;
            document.getElementById('editingProductId').value = id;

            const preview = document.getElementById('imagePreview');
            if (product.imageData) {
                preview.src = product.imageData;
            } else if (product.image) {
                preview.src = product.image;
            } else {
                preview.src = 'https://via.placeholder.com/100/cccccc/666666?text=Sin+imagen';
            }

            document.getElementById('productModal').classList.add('active');
        }

        function saveProduct() {
            const id = document.getElementById('editingProductId').value;
            const title = document.getElementById('productTitle').value.trim();
            const description = document.getElementById('productDescription').value.trim();
            const price = parseFloat(document.getElementById('productPrice').value);
            const image = getProductImage();
            const imageData = document.getElementById('productImageData').value;
            const category = document.getElementById('productCategory').value.trim();
            const stock = parseInt(document.getElementById('productStock').value) || 0;
            const rating = parseFloat(document.getElementById('productRating').value) || 4.5;
            const reviews = parseInt(document.getElementById('productReviews').value) || 0;
            const guarantee = document.getElementById('productGuarantee').checked;
            const guaranteeText = document.getElementById('productGuaranteeText').value.trim();
            const visible = document.getElementById('productVisible').checked;

            if (!title || !description || isNaN(price) || price <= 0) {
                showToast('⚠️ Completa todos los campos correctamente');
                return;
            }

            if (id) {
                const index = products.findIndex(p => p.id === parseInt(id));
                if (index !== -1) {
                    products[index] = {
                        ...products[index],
                        title,
                        description,
                        price,
                        image,
                        imageData: imageData || null,
                        category,
                        stock,
                        rating,
                        reviews,
                        guarantee,
                        guaranteeText: guarantee ? guaranteeText : '',
                        visible
                    };
                    showToast('✅ Producto actualizado correctamente');
                }
            } else {
                const newProduct = {
                    id: nextId++,
                    title,
                    description,
                    price,
                    image,
                    imageData: imageData || null,
                    category,
                    stock,
                    rating,
                    reviews,
                    guarantee,
                    guaranteeText: guarantee ? guaranteeText : '',
                    visible
                };
                products.push(newProduct);
                showToast('✅ Producto agregado correctamente');
            }

            closeProductModal();
            renderAdminTable();
            renderProducts();
        }

        function deleteProduct(id) {
            if (!isAdmin) return;
            if (confirm('¿Estás seguro de eliminar este producto?')) {
                products = products.filter(p => p.id !== id);
                renderAdminTable();
                renderProducts();
                showToast('🗑️ Producto eliminado');
            }
        }

        function editFaq(index) {
            if (!isAdmin) {

        function saveFaq() {
            const index = parseInt(document.getElementById('editingFaqIndex').value);
            const title = document.getElementById('faqTitle').value.trim();
            const content = document.getElementById('faqContent').value.trim();

            if (!title || !content) {
                showToast('⚠️ Completa todos los campos');
                return;
            }

            if (index >= 0 && index < faqItems.length) {
                faqItems[index] = { title, content };
                showToast('✅ FAQ actualizada correctamente');
            } else {
                faqItems.push({ title, content });
                showToast('✅ FAQ agregada correctamente');
            }

            closeFaqModal();
            renderFAQ();
        }
